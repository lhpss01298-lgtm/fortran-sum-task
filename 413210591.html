AI 輸出之 Fortran 原始碼 (sum_n.f90)：

PROGRAM sum_one_to_n
    IMPLICIT NONE
    INTEGER :: n, i
    INTEGER(8) :: total_sum ! 使用 8 位元整數防範溢位
    CHARACTER(LEN=100) :: arg

    ! 檢查是否有命令列參數傳入，預設 N = 100
    IF (COMMAND_ARGUMENT_COUNT() > 0) THEN
        CALL GET_COMMAND_ARGUMENT(1, arg)
        READ(arg, *) n
    ELSE
        n = 100
    END IF

    total_sum = 0
    DO i = 1, n
        total_sum = total_sum + i
    END DO

    PRINT *, '========================================'
    PRINT *, '       Fortran 1 加到 N 計算程式        '
    PRINT *, '========================================'
    PRINT *, '輸入的 N 值為: ', n
    PRINT *, '計算結果 (1 到 ', n, ' 的總和): ', total_sum
    PRINT *, '========================================'
END PROGRAM sum_one_to_n

PROGRAM sum_one_to_n
    IMPLICIT NONE                             ! 強制所有變數必須宣告型態，避免隱含型態錯誤
    INTEGER :: n, i                           ! 宣告整數變數 n (上限值) 與 i (迴圈計數器)
    INTEGER(8) :: total_sum                   ! 宣告 8 位元整數總和變數，防範數值過大溢位
    CHARACTER(LEN=100) :: arg                 ! 宣告字元陣列用以接收命令列傳入的參數

    ! 檢查是否有命令列參數傳入，若無則給定預設值 100
    IF (COMMAND_ARGUMENT_COUNT() > 0) THEN
        CALL GET_COMMAND_ARGUMENT(1, arg)     ! 取得第一個命令列參數
        READ(arg, *) n                        ! 將字串參數轉換為整數賦值給 n
    ELSE
        n = 100                               ! 預設上限值為 100
    END IF

    total_sum = 0                             ! 初始化總和變數為 0
    DO i = 1, n                               ! 迴圈從 1 遞增到 n
        total_sum = total_sum + i             ! 累加數值到總和變數中
    END DO

    PRINT *, '========================================'
    PRINT *, '       Fortran 1 加到 N 計算程式        '
    PRINT *, '========================================'
    PRINT *, '輸入的 N 值為: ', n
    PRINT *, '計算結果 (1 到 ', n, ' 的總和): ', total_sum
    PRINT *, '========================================'
END PROGRAM sum_one_to_n
